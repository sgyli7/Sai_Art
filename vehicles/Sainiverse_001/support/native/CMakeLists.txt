cmake_minimum_required(VERSION 3.22)
project(leviathan_track_path LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GODOT_CPP_PATH "" CACHE PATH "Path to a godot-cpp checkout")
set(GODOTCPP_TARGET template_debug CACHE STRING "")
set(GODOTCPP_PRECISION single CACHE STRING "")
set(GODOTCPP_CUSTOM_API_FILE "${CMAKE_CURRENT_SOURCE_DIR}/api/extension_api.json" CACHE FILEPATH "")
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE FILEPATH "")
add_subdirectory("${GODOT_CPP_PATH}" godot-cpp)
add_library(leviathan_track_path SHARED track_path.cpp)
target_link_libraries(leviathan_track_path PRIVATE godot-cpp)
target_compile_options(leviathan_track_path PRIVATE -O3 -ffp-contract=off)
set_target_properties(leviathan_track_path PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
